# Public pages are crawlable. Private areas are listed so crawlers do not spend time on them; those pages also send
# noindex, and sign-in is always enforced by the application itself (robots.txt is not a security control).
#
# One group on purpose: a crawler that matches a named group (for example Googlebot) ignores the "*" group, so a
# separate "Allow: /" group per bot would bypass these rules.
# Each private area is blocked as an exact path ("$") and as a folder ("/"), so a public page whose address merely
# starts with the same letters (for example /my-story) is never caught.
User-agent: *
Disallow: /admin$
Disallow: /admin/
Disallow: /staff$
Disallow: /staff/
Disallow: /my$
Disallow: /my/
Disallow: /account$
Disallow: /account/
Disallow: /auth$
Disallow: /auth/
Disallow: /reset-password$
Disallow: /reset-password/
Disallow: /lease-to-own/apply
Disallow: /lease-to-own/portal
Allow: /
